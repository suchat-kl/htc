import 'package:dio/dio.dart';
import 'package:flutter/foundation.dart';
import 'api_service.dart';

class AuthInterceptor extends Interceptor {
  final ApiService apiService;

  bool _isRefreshing = false;
  final _pendingRequests =
      <({RequestOptions options, ErrorInterceptorHandler handler})>[];

  AuthInterceptor(this.apiService);

  @override
  void onRequest(
    RequestOptions options,
    RequestInterceptorHandler handler,
  ) async {
    final path = options.path;
    // List of endpoints that don't need token
    final noAuthEndpoints = [
      '/api/auth/login',
      '/api/auth/refresh-token',
      '/api/auth/ticker-msg',
    ];

    // Check if current path needs authentication
    // Check if path is EXACTLY in the list (not partial match)
    final needsAuth = !noAuthEndpoints.contains(path);
    // final needsAuth = !noAuthEndpoints.any(
    //   (endpoint) => options.path.contains(endpoint),
    // );

    if (needsAuth) {
      final token = await apiService.getAccessToken();
      if (token != null && token.isNotEmpty) {
        options.headers['Authorization'] = 'Bearer $token';
        debugPrint('🔑 Added token to request: ${options.path}');
      }
    } else {
      debugPrint('🔓 No auth needed for: ${options.path}');
    }

    handler.next(options);
  }

  @override
  void onError(DioException err, ErrorInterceptorHandler handler) async {
    // Handle both 401 and 403 (some servers return 403 for expired tokens) || err.response?.statusCode == 403
    if (err.response?.statusCode == 401) {
      final requestPath = err.requestOptions.path;

      // Don't try to refresh for auth endpoints
      if (requestPath.contains('/api/auth/login') ||
          requestPath.contains('/api/auth/ticker-msg') ||
          requestPath.contains('/api/auth/refresh-token') ||
          requestPath.contains('/api/auth/logout')) {
        // Added logout
        debugPrint('⚠️ Auth endpoint failed: $requestPath');
        return handler.next(err);
      }

      debugPrint(
        '🔄 401 detected on $requestPath, attempting token refresh...',
      );

      if (!_isRefreshing) {
        _isRefreshing = true;

        try {
          await apiService.refreshToken();
          _isRefreshing = false;

          // Retry pending requests
          for (final pending in _pendingRequests) {
            final newToken = await apiService.getAccessToken();
            pending.options.headers['Authorization'] = 'Bearer $newToken';
            _retryRequest(pending.options, pending.handler);
          }
          _pendingRequests.clear();

          // Retry current request
          final newToken = await apiService.getAccessToken();
          err.requestOptions.headers['Authorization'] = 'Bearer $newToken';
          _retryRequest(err.requestOptions, handler);
        } catch (e) {
          debugPrint('🔴 Token refresh failed: $e');
          _isRefreshing = false;

          for (final pending in _pendingRequests) {
            pending.handler.reject(err);
          }
          _pendingRequests.clear();

          handler.reject(err);
        }
      } else {
        debugPrint('⏳ Refresh in progress, queuing request: $requestPath');
        _pendingRequests.add((options: err.requestOptions, handler: handler));
      }
    } else {
      handler.next(err);
    }
  }

  void _retryRequest(RequestOptions options, ErrorInterceptorHandler handler) {
    debugPrint('🔄 Retrying request: ${options.path}');
    apiService.dio
        .fetch(options)
        .then(
          (response) {
            debugPrint('✅ Retry successful: ${options.path}');
            handler.resolve(response);
          },
          onError: (e) {
            debugPrint('❌ Retry failed: ${options.path}');
            if (e is DioException) {
              handler.reject(e);
            } else {
              handler.reject(DioException(requestOptions: options, error: e));
            }
          },
        );
  }
}
