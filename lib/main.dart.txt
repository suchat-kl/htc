/// ============================ MODEL ============================
enum PermitType { training, otherAgency, small }

extension PermitTypeX on PermitType {
  String get label => switch (this) {
        PermitType.training => 'กองฝึก กรมทางหลวง',
        PermitType.otherAgency => 'หน่วยราชการอื่น',
        PermitType.small => 'รายย่อย (ไม่เกิน 10 ห้อง)',
      };
}

/// ============================ FORM SCREEN ============================
class BookingFormScreen extends StatefulWidget {
  const BookingFormScreen({super.key});

  @override
  State<BookingFormScreen> createState() => _BookingFormScreenState();
}

class _BookingFormScreenState extends State<BookingFormScreen> {
  final _formKey = GlobalKey<FormState>();

  // ---------- Controllers ----------
  final _permitNo = TextEditingController(text: '1905');
  final _bookerName = TextEditingController(text: 'อุทัยรัตน์ เนียรเจริญสุข');
  final _phone = TextEditingController(text: '080 999 5974');
  final _guestCount = TextEditingController(text: '80');
  final _staffCount = TextEditingController(text: '10');
  final _remark = TextEditingController();
  final _orgName = TextEditingController(text: 'กลุ่มพัฒนาทรัพยากรบุคคล');
  final _courseName =
      TextEditingController(text: 'หลักสูตร หัวหน้าหมวดทางหลวงที่ได้รับการแต่งตั้งใหม่');
  final _receiptAddress = TextEditingController();

  // ---------- State ----------
  DateTime bookedDate = DateTime(2026, 7, 8);
  DateTime? startDate = DateTime(2026, 8, 18);
  DateTime? endDate = DateTime(2026, 8, 21);
  String branch = 'ศรีราชา';
  PermitType permitType = PermitType.training;
  bool roomStay = true;
  bool activityRoom = true;

  @override
  void dispose() {
    for (final c in [
      _permitNo, _bookerName, _phone, _guestCount,
      _staffCount, _remark, _orgName, _courseName, _receiptAddress
    ]) {
      c.dispose();
    }
    super.dispose();
  }

  // ---------- Helpers ----------
  String _fmt(DateTime? d) => d == null
      ? ''
      : '${d.day.toString().padLeft(2, '0')}/'
        '${d.month.toString().padLeft(2, '0')}/'
        '${d.year + 543}';

  Future<void> _pickDate({required bool isStart}) async {
    final initial = (isStart ? startDate : endDate) ?? DateTime.now();
    final picked = await showDatePicker(
      context: context,
      initialDate: initial,
      firstDate: DateTime(2020),
      lastDate: DateTime(2035),
      helpText: isStart ? 'เลือกวันที่เริ่มต้น' : 'เลือกวันที่สิ้นสุด',
      builder: (ctx, child) => Theme(
        data: Theme.of(ctx).copyWith(
          colorScheme: const ColorScheme.light(primary: AppColors.primary),
        ),
        child: child!,
      ),
    );
    if (picked == null) return;
    setState(() {
      if (isStart) {
        startDate = picked;
        if (endDate != null && endDate!.isBefore(picked)) endDate = picked;
      } else {
        endDate = picked;
      }
    });
  }

  void _save() {
    if (!_formKey.currentState!.validate()) {
      _toast('กรุณากรอกข้อมูลให้ครบถ้วน', AppColors.danger);
      return;
    }
    if (!roomStay && !activityRoom) {
      _toast('กรุณาเลือกอย่างน้อย 1 รายการ (ห้องพัก / ห้องกิจกรรม)', AppColors.danger);
      return;
    }
    _toast('บันทึกข้อมูลเลขที่ ${_permitNo.text} เรียบร้อยแล้ว', AppColors.success);
  }

  void _toast(String msg, Color color) {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(SnackBar(
        content: Text(msg, style: const TextStyle(fontSize: 15)),
        backgroundColor: color,
        behavior: SnackBarBehavior.floating,
        width: 460,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(8)),
      ));
  }

  /// ปุ่ม "ปิด" → ปิดการทำงาน กลับหน้าจอหลัก
  Future<void> _close() async {
    final ok = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
        title: const Text('ยืนยันการปิดหน้าจอ', style: TextStyle(fontWeight: FontWeight.w700)),
        content: const Text('ข้อมูลที่ยังไม่ได้บันทึกจะหายไป ต้องการปิดหน้าจอนี้หรือไม่?',
            style: TextStyle(fontSize: 15.5)),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('ยกเลิก')),
          FilledButton(
            style: FilledButton.styleFrom(backgroundColor: AppColors.danger),
            onPressed: () => Navigator.pop(ctx, true),
            child: const Text('ปิดหน้าจอ'),
          ),
        ],
      ),
    );
    if (ok == true && mounted) {
      if (Navigator.of(context).canPop()) {
        Navigator.of(context).pop(); // กลับหน้าจอหลัก
      } else {
        Navigator.of(context).pushReplacement(
          MaterialPageRoute(builder: (_) => const MainScreen()),
        );
      }
    }
  }

  /// ============================ BUILD ============================
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: LayoutBuilder(
          builder: (context, c) {
            final w = c.maxWidth;
            final bool isWide = w >= 1100;        // 2 คอลัมน์
            final bool isMedium = w >= 760 && w < 1100;
            final double pad = isWide ? 28 : 16;
            final double scale = w >= 1600 ? 1.0 : (w >= 1200 ? 0.96 : 0.92);

            return Column(
              children: [
                _Header(scale: scale, permitNo: _permitNo.text),
                Expanded(
                  child: Center(
                    child: ConstrainedBox(
                      constraints: const BoxConstraints(maxWidth: 1720),
                      child: Padding(
                        padding: EdgeInsets.fromLTRB(pad, pad * .7, pad, 0),
                        child: Form(
                          key: _formKey,
                          child: _FormCard(
                            child: SingleChildScrollView(
                              padding: EdgeInsets.all(isWide ? 28 : 18),
                              child: isWide
                                  ? Row(
                                      crossAxisAlignment: CrossAxisAlignment.start,
                                      children: [
                                        Expanded(flex: 5, child: _leftColumn(scale, isWide)),
                                        const SizedBox(width: 30),
                                        const _VerticalDivider(),
                                        const SizedBox(width: 30),
                                        Expanded(flex: 6, child: _rightColumn(scale, isWide)),
                                      ],
                                    )
                                  : Column(
                                      crossAxisAlignment: CrossAxisAlignment.stretch,
                                      children: [
                                        _leftColumn(scale, isMedium),
                                        const SizedBox(height: 22),
                                        const Divider(height: 1, color: AppColors.border),
                                        const SizedBox(height: 22),
                                        _rightColumn(scale, isMedium),
                                      ],
                                    ),
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),
                ),
                _FooterBar(scale: scale, onSave: _save, onClose: _close),
              ],
            );
          },
        ),
      ),
    );
  }

  /// ---------------- คอลัมน์ซ้าย ----------------
  Widget _leftColumn(double s, bool wide) {
    final labelW = wide ? 168.0 : 150.0;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        _SectionTitle(text: 'ข้อมูลการจอง', icon: Icons.event_note_rounded, scale: s),
        const SizedBox(height: 18),

        _Field(
          label: 'เลขที่ขออนุญาต', labelWidth: labelW, scale: s,
          child: SizedBox(
            width: 150,
            child: _Input(
              controller: _permitNo,
              readOnly: true,
              align: TextAlign.right,
              scale: s,
              filled: AppColors.primarySoft,
            ),
          ),
        ),
        _Field(
          label: 'วันที่จอง', labelWidth: labelW, scale: s,
          child: _ReadonlyText(value: _fmt(bookedDate), scale: s),
        ),
        _Field(
          label: 'สาขา', labelWidth: labelW, scale: s,
          child: _ReadonlyText(value: branch, scale: s),
        ),
        _Field(
          label: 'ชื่อ-สกุล ผู้จอง', labelWidth: labelW, scale: s, required: true,
          child: _Input(
            controller: _bookerName, scale: s,
            hint: 'ระบุชื่อ-นามสกุลผู้จอง',
            validator: (v) => (v == null || v.trim().isEmpty) ? 'กรุณาระบุชื่อผู้จอง' : null,
          ),
        ),
        _Field(
          label: 'เบอร์ติดต่อ', labelWidth: labelW, scale: s, required: true,
          child: _Input(
            controller: _phone, scale: s, hint: '08X XXX XXXX',
            keyboardType: TextInputType.phone,
            formatters: [FilteringTextInputFormatter.allow(RegExp(r'[0-9\s\-]'))],
            validator: (v) => (v == null || v.trim().length < 9) ? 'เบอร์ติดต่อไม่ถูกต้อง' : null,
          ),
        ),
        _Field(
          label: 'วันที่เริ่มต้น', labelWidth: labelW, scale: s, required: true,
          child: _DateInput(
            value: _fmt(startDate), scale: s, onTap: () => _pickDate(isStart: true),
          ),
        ),
        _Field(
          label: 'วันที่สิ้นสุด', labelWidth: labelW, scale: s, required: true,
          child: _DateInput(
            value: _fmt(endDate), scale: s, onTap: () => _pickDate(isStart: false),
          ),
        ),
        _Field(
          label: 'จำนวนผู้เข้าพัก', labelWidth: labelW, scale: s,
          child: Row(children: [
            SizedBox(
              width: 130,
              child: _Input(
                controller: _guestCount, scale: s, align: TextAlign.right,
                keyboardType: TextInputType.number,
                formatters: [FilteringTextInputFormatter.digitsOnly],
              ),
            ),
            SizedBox(width: 10),
            Text('คน', style: TextStyle(fontSize: 15.5 * s, color: AppColors.textSoft)),
          ]),
        ),
        _Field(
          label: 'จำนวนเจ้าหน้าที่', labelWidth: labelW, scale: s,
          child: Row(children: [
            SizedBox(
              width: 130,
              child: _Input(
                controller: _staffCount, scale: s, align: TextAlign.right,
                keyboardType: TextInputType.number,
                formatters: [FilteringTextInputFormatter.digitsOnly],
              ),
            ),
            SizedBox(width: 10),
            Text('คน', style: TextStyle(fontSize: 15.5 * s, color: AppColors.textSoft)),
          ]),
        ),
        const SizedBox(height: 6),
        _Field(
          label: 'ห้องพัก', labelWidth: labelW, scale: s, dense: true,
          child: _CheckRow(
            value: roomStay, scale: s,
            caption: 'ต้องการใช้ห้องพัก',
            onChanged: (v) => setState(() => roomStay = v),
          ),
        ),
        _Field(
          label: 'ห้องกิจกรรม', labelWidth: labelW, scale: s, dense: true,
          child: _CheckRow(
            value: activityRoom, scale: s,
            caption: 'ต้องการใช้ห้องกิจกรรม',
            onChanged: (v) => setState(() => activityRoom = v),
          ),
        ),
      ],
    );
  }

  /// ---------------- คอลัมน์ขวา ----------------
  Widget _rightColumn(double s, bool wide) {
    final labelW = wide ? 200.0 : 160.0;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        _SectionTitle(text: 'รายละเอียดการขออนุญาต', icon: Icons.assignment_turned_in_outlined, scale: s),
        const SizedBox(height: 18),

        _Field(
          label: 'ประเภทการขออนุญาต', labelWidth: labelW, scale: s, alignTop: true,
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: PermitType.values
                .map((t) => _RadioRow(
                      label: t.label,
                      selected: permitType == t,
                      scale: s,
                      onTap: () => setState(() => permitType = t),
                    ))
                .toList(),
          ),
        ),
        const SizedBox(height: 8),
        _Field(
          label: 'หมายเหตุ', labelWidth: labelW, scale: s, alignTop: true,
          child: _Input(
            controller: _remark, scale: s, maxLines: 5,
            hint: 'ระบุรายละเอียดเพิ่มเติม (ถ้ามี)',
          ),
        ),
        const SizedBox(height: 22),

        _SubHeader(text: 'กรณีจองในนามหน่วยงาน', scale: s),
        const SizedBox(height: 16),

        _Field(
          label: 'ชื่อหน่วยงาน', labelWidth: labelW, scale: s,
          child: _Input(controller: _orgName, scale: s, hint: 'ระบุชื่อหน่วยงาน'),
        ),
        _Field(
          label: 'ชื่อหลักสูตร/โครงการ/เรื่อง', labelWidth: labelW, scale: s,
          child: _Input(controller: _courseName, scale: s, hint: 'ระบุชื่อหลักสูตรหรือโครงการ'),
        ),
        _Field(
          label: 'ที่อยู่ในการออกใบเสร็จ', labelWidth: labelW, scale: s, alignTop: true,
          child: _Input(
            controller: _receiptAddress, scale: s, maxLines: 5,
            hint: 'บ้านเลขที่ / ถนน / ตำบล / อำเภอ / จังหวัด / รหัสไปรษณีย์',
          ),
        ),
      ],
    );
  }
}

/// แถวฟอร์ม: label (คงที่) + field (ยืดหยุ่นด้วย Flexible/Expanded)
class _Field extends StatelessWidget {
  final String label;
  final double labelWidth;
  final Widget child;
  final double scale;
  final bool required;
  final bool alignTop;
  final bool dense;

  const _Field({
    required this.label,
    required this.labelWidth,
    required this.child,
    required this.scale,
    this.required = false,
    this.alignTop = false,
    this.dense = false,
  });

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: EdgeInsets.only(bottom: dense ? 8 : 14),
      child: Row(
        crossAxisAlignment:
            alignTop ? CrossAxisAlignment.start : CrossAxisAlignment.center,
        children: [
          SizedBox(
            width: labelWidth,
            child: Padding(
              padding: EdgeInsets.only(top: alignTop ? 12 : 0),
              child: RichText(
                textAlign: TextAlign.right,
                text: TextSpan(
                  text: label,
                  style: TextStyle(
                      fontSize: 15.5 * scale,
                      height: 1.35,
                      fontWeight: FontWeight.w600,
                      color: AppColors.textSoft),
                  children: required
                      ? const [TextSpan(text: ' *', style: TextStyle(color: AppColors.danger))]
                      : null,
                ),
              ),
            ),
          ),
          const SizedBox(width: 14),
          Expanded(child: Align(alignment: Alignment.centerLeft, child: child)),
        ],
      ),
    );
  }
}

class _Input extends StatelessWidget {
  final TextEditingController controller;
  final double scale;
  final String? hint;
  final int maxLines;
  final bool readOnly;
  final TextAlign align;
  final Color? filled;
  final TextInputType? keyboardType;
  final List<TextInputFormatter>? formatters;
  final String? Function(String?)? validator;

  const _Input({
    required this.controller,
    required this.scale,
    this.hint,
    this.maxLines = 1,
    this.readOnly = false,
    this.align = TextAlign.start,
    this.filled,
    this.keyboardType,
    this.formatters,
    this.validator,
  });

  @override
  Widget build(BuildContext context) {
    return TextFormField(
      controller: controller,
      readOnly: readOnly,
      maxLines: maxLines,
      textAlign: align,
      keyboardType: keyboardType,
      inputFormatters: formatters,
      validator: validator,
      style: TextStyle(fontSize: 15.5 * scale, color: AppColors.text, height: 1.4),
      decoration: InputDecoration(
        hintText: hint,
        hintStyle: TextStyle(fontSize: 14.5 * scale, color: const Color(0xFFA0A8B5)),
        isDense: true,
        filled: true,
        fillColor: filled ?? (readOnly ? const Color(0xFFF3F5F9) : Colors.white),
        contentPadding: EdgeInsets.symmetric(horizontal: 14, vertical: maxLines > 1 ? 14 : 13),
        border: _b(AppColors.border),
        enabledBorder: _b(AppColors.border),
        focusedBorder: _b(AppColors.accent, width: 1.6),
        errorBorder: _b(AppColors.danger),
        focusedErrorBorder: _b(AppColors.danger, width: 1.6),
        errorStyle: TextStyle(fontSize: 12.5 * scale),
      ),
    );
  }

  OutlineInputBorder _b(Color c, {double width = 1}) => OutlineInputBorder(
        borderRadius: BorderRadius.circular(8),
        borderSide: BorderSide(color: c, width: width),
      );
}
